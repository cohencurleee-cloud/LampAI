const GROQ = "https://api.groq.com/openai/v1"; const CHAT_MODEL = "qwen/qwen3.6-27b"; const STT_MODEL = "whisper-large-v3-turbo"; const TTS_MODEL = "canopylabs/orpheus-v1-english"; const TTS_VOICE = "troy"; function clean(text = "") { return String(text) .replace(/[\s\S]*?<\/think>/gi, "") .replace(/[\s\S]*/gi, "") .replace(/\*\*/g, "") .replace(/\*/g, "") .trim(); } function stripVoiceTags(text = "") { return clean(text) .replace(/\[[^\]\n]{1,40}\]\s*/g, "") .replace(/\s+/g, " ") .trim(); } async function jsonFrom(response) { const raw = await response.text(); try { return { raw, data: raw ? JSON.parse(raw) : {} }; } catch { return { raw, data: {} }; } } async function callChat( apiKey, messages, maxTokens = 700, temperature = 0.7 ) { const response = await fetch( `${GROQ}/chat/completions`, { method: "POST", headers: { "Content-Type": "application/json", Authorization: `Bearer ${apiKey}` }, body: JSON.stringify({ model: CHAT_MODEL, messages, reasoning_effort: "none", reasoning_format: "hidden", temperature, top_p: 0.85, max_completion_tokens: maxTokens }) } ); const parsed = await jsonFrom(response); if (!response.ok) { throw new Error( parsed.data?.error?.message || parsed.raw || `Groq chat failed (${response.status})` ); } return clean( parsed.data?.choices?.[0]?.message?.content || "" ); } function decodeDataUrl(dataUrl) { const match = /^data:([^;,]+)?(;base64)?,(.*)$/s.exec( dataUrl || "" ); if (!match) { throw new Error("Invalid audio data"); } const mimeType = match[1] || "audio/mp4"; const payload = match[3] || ""; if (!match[2]) { return { mimeType, bytes: new TextEncoder().encode( decodeURIComponent(payload) ) }; } return { mimeType, bytes: new Uint8Array( Buffer.from( payload, "base64" ) ) }; } async function voiceTranscribe( res, apiKey, body ) { try { if (!body.audio) { return res.status(400).json({ error: "No microphone audio received.", stage: "transcribe" }); } const { mimeType, bytes } = decodeDataUrl( body.audio ); if ( bytes.byteLength > 3000000 ) { return res.status(413).json({ error: "Voice turn is too long.", stage: "transcribe" }); } const form = new FormData(); form.append( "file", new Blob( [bytes], { type: mimeType } ), body.filename || "voice.m4a" ); form.append( "model", STT_MODEL ); form.append( "language", "en" ); form.append( "response_format", "json" ); const response = await fetch( `${GROQ}/audio/transcriptions`, { method: "POST", headers: { Authorization: `Bearer ${apiKey}` }, body: form } ); const parsed = await jsonFrom( response ); if (!response.ok) { return res .status( response.status ) .json({ error: parsed.data ?.error ?.message || parsed.raw || "Transcription failed.", stage: "transcribe" }); } const transcript = String( parsed.data?.text || "" ).trim(); if (!transcript) { return res .status(400) .json({ error: "I couldn't hear any words.", stage: "transcribe" }); } return res .status(200) .json({ transcript, stage: "transcribe" }); } catch (error) { return res .status(500) .json({ error: error.message || "Transcription crashed.", stage: "transcribe" }); } } async function voiceReply( res, apiKey, body ) { try { const transcript = String( body.transcript || "" ).trim(); const instructions = String( body.instructions || "" ).trim(); if (!transcript) { return res .status(400) .json({ error: "Missing transcript.", stage: "reply" }); } const history = Array.isArray( body.history ) ? body.history .slice(-10) .filter( x => x && [ "user", "assistant" ].includes( x.role ) && typeof x.content === "string" ) : []; const system = ` You are LampAI in live voice mode. Talk like a real person on a voice call. Keep replies short and conversational. Usually answer in one or two sentences. Never use markdown. Never use bullets. Never use headings. Never use asterisks. Never output tags. Never reveal hidden reasoning. You may occasionally use natural fillers like: uh hmm yeah well A tiny stutter is okay occasionally. Do not overdo it. You may optionally start with one voice direction such as: [warm] [breathy] [whisper] [sarcastic] [exasperated sigh] Most replies should use no direction. ${ instructions ? `User customization:\n${instructions}` : "" } `; const rawReply = await callChat( apiKey, [ { role: "system", content: system }, ...history, { role: "user", content: transcript } ], 120, 0.8 ); const reply = stripVoiceTags( rawReply ); if (!reply) { return res .status(502) .json({ error: "LampAI returned no reply.", stage: "reply" }); } const direction = rawReply.match( /\[[^\]\n]{1,40}\]/ )?.[0] || ""; const speechText = `${ direction ? direction + " " : "" }${reply}` .trim() .slice( 0, 195 ); return res .status(200) .json({ reply, speech_text: speechText, stage: "reply" }); } catch (error) { return res .status(500) .json({ error: error.message || "Reply failed.", stage: "reply" }); } } async function voiceSpeak( res, apiKey, body ) { try { const input = String( body.speech_text || body.text || "" ) .trim() .slice( 0, 195 ); if (!input) { return res .status(400) .json({ error: "Missing speech text.", stage: "speak" }); } const response = await fetch( `${GROQ}/audio/speech`, { method: "POST", headers: { "Content-Type": "application/json", Authorization: `Bearer ${apiKey}` }, body: JSON.stringify({ model: TTS_MODEL, voice: TTS_VOICE, input, response_format: "wav" }) } ); if (!response.ok) { const raw = await response.text(); return res .status(200) .json({ audio: "", warning: raw || "TTS unavailable", stage: "speak" }); } const audio = Buffer.from( await response.arrayBuffer() ).toString( "base64" ); return res .status(200) .json({ audio, stage: "speak" }); } catch (error) { return res .status(200) .json({ audio: "", warning: error.message || "TTS unavailable", stage: "speak" }); } } async function normalChat( res, apiKey, body ) { try { const message = String( body.message || "" ).trim(); const instructions = String( body.instructions || "" ).trim(); const attachments = Array.isArray( body.attachments ) ? body.attachments.slice( 0, 5 ) : []; if ( !message && attachments.length === 0 ) { return res .status(400) .json({ error: "Send a message, image, or file." }); } const content = []; if (message) { content.push({ type: "text", text: message }); } for ( const file of attachments ) { const name = String( file?.name || "attachment" ); const type = String( file?.type || "" ); const value = typeof file?.content === "string" ? file.content : ""; if (!value) { continue; } if ( type.startsWith( "image/" ) && value.startsWith( "data:image/" ) ) { content.push({ type: "image_url", image_url: { url: value } }); continue; } if ( file.encoding === "text" || type.startsWith( "text/" ) || /\.(txt|md|csv|json)$/i.test( name ) ) { content.push({ type: "text", text: `Attached file: ${name}\n\n${value}` }); continue; } content.push({ type: "text", text: `The user attached ${name}, but this file type cannot currently be read.` }); } const system = ` You are LampAI. Answer naturally and directly. Usually respond in 1 to 4 short sentences. Never reveal private reasoning. Never output tags. Do not use markdown asterisks. Do not add unnecessary headings. ${ instructions ? `User customization:\n${instructions}` : "" } `; const reply = await callChat( apiKey, [ { role: "system", content: system }, { role: "user", content } ] ); if (!reply) { return res .status(502) .json({ error: "Groq returned no response." }); } return res .status(200) .json({ reply }); } catch (error) { return res .status(500) .json({ error: error.message || "Chat failed." }); } } export default async function handler( req, res ) { res.setHeader( "Access-Control-Allow-Origin", "*" ); res.setHeader( "Access-Control-Allow-Methods", "POST, OPTIONS" ); res.setHeader( "Access-Control-Allow-Headers", "Content-Type" ); if ( req.method === "OPTIONS" ) { return res .status(204) .end(); } if ( req.method !== "POST" ) { return res .status(405) .json({ error: "Method not allowed" }); } try { const apiKey = process.env .GROQ_API_KEY; if (!apiKey) { return res .status(500) .json({ error: "Missing GROQ_API_KEY in Vercel." }); } const body = typeof req.body === "string" ? JSON.parse( req.body ) : ( req.body || {} ); if ( body.action === "voice_transcribe" ) { return voiceTranscribe( res, apiKey, body ); } if ( body.action === "voice_reply" ) { return voiceReply( res, apiKey, body ); } if ( body.action === "voice_speak" ) { return voiceSpeak( res, apiKey, body ); } return normalChat( res, apiKey, body ); } catch (error) { console.error( "LampAI fatal error:", error ); return res .status(500) .json({ error: error.message || "Server error." }); } }
